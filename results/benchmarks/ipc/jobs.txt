# Jobs for IPC


# 1. Precision 

mkdir -p results/benchmarks/ipc/precision

# Add more hamiltonians/change param range if you want
for hamiltonian in random native-exchange; do
    mkdir -p "results/benchmarks/ipc/precision/${hamiltonian}"

    for backend in shots; do
        for precision in 0.01 0.05 0.1; do
            for seed in $(seq 1 50); do
                echo "Running hamiltonian=$hamiltonian backend=$backend precision=$precision seed=$seed"

                python -m experiments.run_benchmarks \
                    --tasks ipc \
                    --backend "$backend" \
                    --hamiltonian "$hamiltonian" \
                    --precision "$precision" \
                    --seed "$seed" \
                    > "results/benchmarks/ipc/precision/${hamiltonian}/${backend}_precision${precision}_seed${seed}.txt" 2>&1
            done
        done
    done
done

# 2. Depth

mkdir -p results/benchmarks/ipc/depth

for hamiltonian in random native-exchange; do
    mkdir -p "results/benchmarks/ipc/depth/${hamiltonian}"

    for depth in 1 3 8 15; do
        seq 1 10 | xargs -P 2 -I {} bash -c '
            hamiltonian="$1"
            depth="$2"
            seed="$3"

            echo "Running hamiltonian=$hamiltonian depth=$depth backend=shots precision=0.01 seed=$seed"

            python -m experiments.run_benchmarks \
                --tasks ipc \
                --backend shots \
                --hamiltonian "$hamiltonian" \
                --precision 0.01 \
                --seed "$seed" \
                --ipc-reservoir-depth "$depth" \
                > "results/benchmarks/ipc/depth/${hamiltonian}/shots_precision0.01_depth${depth}_seed${seed}.txt" 2>&1
        ' _ "$hamiltonian" "$depth" {}
    done
done

                --precision 0.01 \

                --ipc-window-length "$window_length" \
                --max-delay "$max_delay" \
                > "results/benchmarks/ipc/window/${hamiltonian}/shots_precision0.01_window${window_length}_seed${seed}.txt" 2>&1
        ' _ "$hamiltonian" "$window_length" {}
    done
done
